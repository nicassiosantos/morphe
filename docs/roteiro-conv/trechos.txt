[5] Parte A --- desenvolver o módulo / A2 Da tabela a dois laços
========================================================================
para n = 0 até nx + nh - 2:             # uma LINHA da tabela
    acc = 0
    para k = kmin(n) até kmax(n):       # as casas coloridas
        acc = acc + x[k] * h[n - k]
    y[n] = acc                          # a soma da linha

[6] Parte A --- desenvolver o módulo / A3 O que entra e o que sai do módulo

module conv_aluno #(
    parameter DATA_WIDTH  = 32,
    parameter X_ADDR_BITS = 10,     // x: ate 1024 amostras
    parameter H_ADDR_BITS = 10,     // h: ate 1024 amostras
    parameter Y_ADDR_BITS = 11      // y: ate 2047 amostras (nx + nh - 1)
)(
    input  wire                          clk,       // 50 MHz
    input  wire                          reset_n,   // ativo em 0

    // --- controle (PIOs) ---
    input  wire                          start,     // HPS -> FPGA
    output reg                           done,      // FPGA -> HPS
    input  wire [X_ADDR_BITS:0]          nx,        // amostras de x, 1 a 1024
    input  wire [H_ADDR_BITS:0]          nh,        // amostras de h, 1 a 1024

    // --- memoria de x[k] (so leitura) ---
    output reg  [X_ADDR_BITS-1:0]        x_addr,
    input  wire signed [DATA_WIDTH-1:0]  x_data,

    // --- memoria de h[k] (so leitura) ---
    output reg  [H_ADDR_BITS-1:0]        h_addr,
    input  wire signed [DATA_WIDTH-1:0]  h_data,

    // --- memoria de y[n] (so escrita) ---
    output reg  [Y_ADDR_BITS-1:0]        y_addr,
    output reg  signed [DATA_WIDTH-1:0]  y_data,
    output reg                           y_we
);

    // nx e nh com sinal, na largura de n e k (13 bits)
    wire signed [12:0] nx_s = $signed({2'b00, nx});
    wire signed [12:0] nh_s = $signed({2'b00, nh});

    // kmin = max(0, n - nh + 1)        kmax = min(n, nx - 1)
    wire signed [12:0] kmin = (n - nh_s + 1 > 0) ? n - nh_s + 1 : 0;
    wire signed [12:0] kmax = (n < nx_s - 1)     ? n : nx_s - 1;

[8] Parte A --- desenvolver o módulo / A5 Escrever o Verilog, por partes

    // os estados
    localparam [2:0] S_OCIOSO     = 3'd0;
    localparam [2:0] S_LINHA      = 3'd1;
    localparam [2:0] S_ENDERECO   = 3'd2;
    localparam [2:0] S_ESPERA     = 3'd3;
    localparam [2:0] S_MULTIPLICA = 3'd4;
    localparam [2:0] S_ACUMULA    = 3'd5;
    localparam [2:0] S_GRAVA      = 3'd6;
    localparam [2:0] S_FIM        = 3'd7;

    reg [2:0] estado;

    // borda de subida do start: 1 so no ciclo em que ele passa de 0 a 1
    reg  start_d;
    wire start_pulso = start & ~start_d;
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) start_d <= 1'b0;
        else          start_d <= start;
    end

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            estado <= S_OCIOSO;
            done   <= 1'b0;
            y_we   <= 1'b0;
            // zere aqui tambem os seus registradores
        end else begin
            y_we <= 1'b0;           // so fica em 1 no ciclo de GRAVA

            case (estado)
                S_OCIOSO: begin
                    done <= 1'b0;
                    if (start_pulso) begin
                        n      <= 0;
                        estado <= S_LINHA;
                    end
                end

                // S_LINHA, S_ENDERECO, S_ESPERA, S_MULTIPLICA,
                // S_ACUMULA e S_GRAVA: escreva aqui

                S_FIM: begin
                    done <= 1'b1;   // fica em 1 ate o HPS baixar o start
                    if (!start) estado <= S_OCIOSO;
                end

                default: estado <= S_OCIOSO;
            endcase
        end
    end
